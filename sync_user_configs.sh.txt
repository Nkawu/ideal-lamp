#!/bin/bash

# Destination directory for YAML files
DEST_DIR="${DEST_DIR:-/nvme1/utils/automation/ansible/users}"

# Create destination directory if it doesn't exist
mkdir -p "$DEST_DIR"

# Parse /etc/passwd file
while IFS=: read -r username password uid gid gecos home shell; do
    # Skip system users (UID < 1000), users without a home directory, nologin users, and gitlab users
    if [[ $uid -lt 1000 ]] || [[ ! -d "$home" ]] || [[ "$shell" == "/usr/sbin/nologin" ]] || [[ "$username" == *gitlab* ]]; then
        continue
    fi

    # Define YAML file path
    yaml_file="$DEST_DIR/${username}.yml"

    # Skip if YAML file already exists
    if [[ -f "$yaml_file" ]]; then
        echo "Skipping $username - YAML file already exists"
        continue
    fi

    echo "Creating YAML config for user: $username"

    # Extract full name from GECOS field (before first comma)
    full_name=$(echo "$gecos" | cut -d',' -f1)

    # Initialize YAML content
    cat > "$yaml_file" <<EOF
---
user: $username
full_name: $full_name
EOF

    # Process SSH authorized_keys
    authorized_keys_file="$home/.ssh/authorized_keys"
    if [[ -f "$authorized_keys_file" ]]; then
        # Read all keys into an array
        mapfile -t keys < "$authorized_keys_file"

        # Filter out empty lines
        non_empty_keys=()
        for key in "${keys[@]}"; do
            [[ -n "$key" ]] && non_empty_keys+=("$key")
        done

        total_keys=${#non_empty_keys[@]}

        # Process each key
        for i in "${!non_empty_keys[@]}"; do
            key="${non_empty_keys[$i]}"

            # Comment out all keys except the last non-empty one
            if [[ $((i + 1)) -eq $total_keys ]]; then
                echo "public_key: $key" >> "$yaml_file"
            else
                echo "# public_key: $key" >> "$yaml_file"
            fi
        done
    else
        echo "public_key: null" >> "$yaml_file"
    fi

    # Extract VNC port from vnc-start.sh
    vnc_start_file="$home/vnc-start.sh"
    if [[ -f "$vnc_start_file" ]]; then
        # Extract the :XX port number from vncserver command
        vnc_port=$(grep 'vncserver' "$vnc_start_file" | sed -n 's/.*:\([0-9][0-9]*\).*/\1/p' | head -n 1)
        if [[ -n "$vnc_port" ]]; then
            echo "vnc_port: $vnc_port" >> "$yaml_file"
        else
            echo "vnc_port: null" >> "$yaml_file"
        fi
    else
        echo "vnc_port: null" >> "$yaml_file"
    fi

    echo "Created: $yaml_file"

done < /etc/passwd

echo "YAML config sync complete!"
